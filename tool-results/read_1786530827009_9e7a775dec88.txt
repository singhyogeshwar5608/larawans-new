    24→function toDurationCategory(duration: string): string {
    25→  const weeks = parseInt(duration, 10);
    26→  if (Number.isNaN(weeks)) return "12+ Weeks";
    27→  if (weeks <= 4) return "1-4 Weeks";
    28→  if (weeks <= 8) return "4-8 Weeks";
    29→  if (weeks <= 12) return "8-12 Weeks";
    30→  return "12+ Weeks";
    31→}
    32→
    33→function toPriceCategory(price: string): string {
    34→  const amount = parseInt(price.replace(/[^\d]/g, ""), 10);
    35→  if (Number.isNaN(amount)) return "₹5,000+";
    36→  if (amount === 0) return "Free";
    37→  if (amount <= 999) return "₹0 - ₹999";
    38→  if (amount <= 4999) return "₹1,000 - ₹4,999";
    39→  return "₹5,000+";
    40→}
